    \section{Syntax}

    \subsection{Types}

    \[
        \begin{array}{rcl}
            A,B & ::= & \mathsf{Base} \\
            & \mid & A \rightarrow B
        \end{array}
    \]

    where

    \begin{itemize}
        \item $\mathsf{Base}$ is the base type.
        \item $A \rightarrow B$ is the function type.
    \end{itemize}

    \subsection{Terms}

    \[
        \begin{array}{rcl}
            t & ::= & x \\
            & \mid & \lambda.t \\
            & \mid & t\;t
        \end{array}
    \]

    Variables are represented by de Bruijn indices.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
    \section{Typing Rules}

    Typing judgment

    \[
        \Gamma \vdash t : A
    \]

    where

    \begin{itemize}
        \item $\Gamma$ is a typing context.
        \item $\mathrm{Lookup}(\Gamma,x,A)$ denotes that
        variable $x$ has type $A$ in $\Gamma$.
    \end{itemize}

    \begin{mathpar}

        \inferrule*[right=T-Var]
        {\mathrm{Lookup}(\Gamma,x,A)}
        {\Gamma \vdash x : A}

        \inferrule*[right=T-Lam]
        {\Gamma,A \vdash t : B}
        {\Gamma \vdash \lambda.t : A \rightarrow B}

        \inferrule*[right=T-App]
        {\Gamma \vdash t : A \rightarrow B \\
        \Gamma \vdash u : A}
        {\Gamma \vdash t\,u : B}

    \end{mathpar}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
    \section{Operational Semantics}

    Evaluation relation

    \[
        t \longrightarrow t'
    \]

    Values

    \[
        \begin{array}{rcl}
            v & ::= & \lambda.t
        \end{array}
    \]

    \begin{mathpar}

        \inferrule*[right=E-AppAbs]
        {}
        {(\lambda.t)\;v \longrightarrow t[v/0]}

        \inferrule*[right=E-App1]
        {t_1 \longrightarrow t_1'}
        {t_1\;t_2 \longrightarrow t_1'\;t_2}

        \inferrule*[right=E-App2]
        {t_2 \longrightarrow t_2'}
        {v\;t_2 \longrightarrow v\;t_2'}

    \end{mathpar}

    Here, $t[v/0]$ denotes substitution of the value $v$
    for the variable with de Bruijn index $0$.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
    \section{Metatheory}

    The Lean formalization proves the following lemmas and
    theorems.

    \begin{itemize}
        \item Rename Lemma
        \item Substitution Lemma
        \item Progress Theorem
        \item Preservation Theorem
        \item Type Soundness Theorem
    \end{itemize}

    The central result is

    \[
        \text{Well-typed programs do not get stuck.}
    \]

    \section{Correspondence with the Lean Implementation}

    \begin{center}
        \begin{tabular}{ll}
            \toprule
            Lean implementation & Mathematical notation \\
            \midrule
            \texttt{Ty.base} & $\mathsf{Base}$ \\
            \texttt{Ty.arr A B} & $A \rightarrow B$ \\
            \texttt{Lookup} & $\mathrm{Lookup}(\Gamma,x,A)$ \\
            \texttt{Term.var} & $x$ \\
            \texttt{Term.lam} & $\lambda.t$ \\
            \texttt{Term.ap} & $t\,u$ \\
            \bottomrule
        \end{tabular}
    \end{center}

    The typing context is represented as a list of types, and
    variable lookup is defined as an inductive relation rather
    than as a partial function.
